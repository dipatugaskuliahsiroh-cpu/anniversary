// --- Komponen: Love Counter ---
const LoveCounter = ({ targetDate }: { targetDate: string }) => {
  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 });

  useEffect(() => {
    const interval = setInterval(() => {
      const start = new Date(targetDate).getTime();
      const now = new Date().getTime();
      const diff = now - start;

      if (diff > 0) {
        setTimeLeft({
          days: Math.floor(diff / (1000 * 60 * 60 * 24)),
          hours: Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)),
          minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
          seconds: Math.floor((diff % (1000 * 60)) / 1000),
        });
      }
    }, 1000);
    return () => clearInterval(interval);
  }, [targetDate]);

  return (
    <div className="flex gap-4 justify-center my-8">
      {Object.entries(timeLeft).map(([unit, value]) => (
        <div key={unit} className="flex flex-col items-center bg-white/80 backdrop-blur-sm p-3 rounded-xl shadow-lg w-20 border-2 border-rose-100">
          <span className="text-2xl font-bold text-rose-500">{String(value).padStart(2, "0")}</span>
          <span className="text-xs uppercase text-gray-500 font-medium">{unit}</span>
        </div>
      ))}
    </div>
  );
};

// --- Komponen: Music Player ---
const MusicPlayer = ({ url }: { url: string }) => {
  const [isPlaying, setIsPlaying] = useState(false);
  const audioRef = useRef<HTMLAudioElement>(null);

  const toggle = () => {
    if (audioRef.current) {
      if (isPlaying) audioRef.current.pause();
      else audioRef.current.play();
      setIsPlaying(!isPlaying);
    }
  };

  return (
    <div className="fixed bottom-6 right-6 z-50">
      <motion.div 
        whileHover={{ scale: 1.1 }} 
        whileTap={{ scale: 0.9 }}
        className="bg-white/90 p-4 rounded-full shadow-2xl cursor-pointer border-2 border-rose-400"
        onClick={toggle}
      >
        {isPlaying ? <Pause className="text-rose-500 w-6 h-6" /> : <Music className="text-rose-500 w-6 h-6" />}
      </motion.div>
      <audio ref={audioRef} src={url} loop />
    </div>
  );
};
